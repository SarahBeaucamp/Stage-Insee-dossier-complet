db = DuckDBClient.of({})

// 2. Connexion à la base 2023 contenant tes prédictions ET tes SHAP values
chemin_s3 = 'https://minio.lab.sspcloud.fr/sarahbeaucamp/predictions_xgboost_2023.parquet'
isGeoSelected = (selection_geo !== null);

htl.html`
  <div style="font-size: 1.2em; margin-bottom: 10px;">
    <strong>Commune analysée :</strong> ${isGeoSelected ? selection_geo.GEO_LABEL : "Veuillez sélectionner un territoire à gauche"} 
    ${isGeoSelected ? `<span style="color: gray;">(${selection_geo.GEO})</span>` : ""}
  </div>
`
// Requête DuckDB MODIFIÉE : On extrait Y_OBSERVE, Y_PREDIT et le Top 5 SHAP
resultat_commune = isGeoSelected ? 
  db.queryRow(`
    SELECT Y_OBSERVE, Y_PREDIT_XGB, 
           VAR_1, SHAP_1, VAR_2, SHAP_2, VAR_3, SHAP_3, VAR_4, SHAP_4, VAR_5, SHAP_5
    FROM read_parquet('${chemin_s3}')
    WHERE GEO = '${selection_geo.GEO}'
  `) : null;
// Création d'encarts HTML dynamiques (KPIs)
htl.html`
  ${isGeoSelected && resultat_commune ? htl.html`
    <div class="kpi-container">
      <div class="kpi-card obs-card">
        <p class="kpi-title">Valeur Réelle Observée (2023)</p>
        <p class="kpi-value obs-value">${(resultat_commune.Y_OBSERVE * 100).toFixed(2)} %</p>
      </div>
      
      <div class="kpi-card pred-card">
        <p class="kpi-title">Prédiction du Modèle XGBoost</p>
        <p class="kpi-value pred-value">${(resultat_commune.Y_PREDIT_XGB * 100).toFixed(2)} %</p>
      </div>
    </div>
  ` : htl.html`<p style="color: gray; font-style: italic;">"Par souci de significativité statistique, les résultats ne sont pas présentés pour les communes de moins de 500 habitants."</p>`}
`
data_shap = (isGeoSelected && resultat_commune && resultat_commune.VAR_1) ? [
  { variable: resultat_commune.VAR_1, impact: Number(resultat_commune.SHAP_1) },
  { variable: resultat_commune.VAR_2, impact: Number(resultat_commune.SHAP_2) },
  { variable: resultat_commune.VAR_3, impact: Number(resultat_commune.SHAP_3) },
  { variable: resultat_commune.VAR_4, impact: Number(resultat_commune.SHAP_4) },
  { variable: resultat_commune.VAR_5, impact: Number(resultat_commune.SHAP_5) }
] : [];

// On ajoute une condition : si les données sont vides, on affiche un message propre au lieu d'un bug
data_shap.length > 0 ? Plot.plot({
  title: `Top 5 des variables expliquant la prédiction pour ${selection_geo.GEO_LABEL}`,
  subtitle: "Impact local sur l'écart d'activité (en points)",
  height: 280,
  marginLeft: 320, 
  x: { 
    label: "Impact (Valeur SHAP)", 
    grid: true 
  },
  y: { 
    label: null,
    domain: data_shap.map(d => d.variable).reverse()
  },
  marks: [
    Plot.barX(data_shap, {
      x: "impact",
      y: "variable",
      fill: d => d.impact > 0 ? "Pousse vers les Hommes (+)" : "Pousse vers les Femmes (-)",
      rx: 3
    }),
    Plot.ruleX([0], { stroke: "#2c3e50", strokeWidth: 2 })
  ],
  color: { 
    domain: ["Pousse vers les Hommes (+)", "Pousse vers les Femmes (-)"], 
    range: ["#e67e22", "#2980b9"],
    legend: true 
  }
}) : htl.html`<p style="color: gray; font-style: italic; padding: 20px;">"Par souci de significativité statistique, les résultats ne sont pas présentés pour les communes de moins de 500 habitants."</p>`
